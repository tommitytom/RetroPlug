# retroplug-export-plan (main.cpp): the plan exporter's build-time tool (docs/modular-graph-phase-e3-design.md section
# 4). A device image's build runs it over the image's plan.json and embeds the header it writes, so a plan the device
# cannot run, or one its arena cannot hold, fails the firmware's build rather than its boot. The root CMakeLists adds
# this directory once the packages it lists are added.
#
# A device product of the desktop build (cmake/packages.cmake), so it registers the device entries of the packages it
# lists as the desktop builds them, which plan; beside them, the same entries as the device builds them
# (retroplug_device_mode_objects), which the loader's sizing pass sizes the plan with. Not shipped, and built by name:
# the scripts that build a device image build it first, and retroplug-device-mode-test runs it.
#
# The list is every package a device image's plan may name: the box's (core, device-io, n8) and the device check's,
# which adds dsp and devtest (rp.devtest.plan, test types no shipped product carries, for its plan's coupled pair). An
# image's build passes its own product's list (--packages), and the tool registers only those, both builds: a plan
# naming a type its image does not register is refused here, and the firmware's build with it, rather than at boot.
set(RP_EXPORT_PLAN_PACKAGES core dsp n8 device-io devtest)

# --packages' table: each listed package's device entries, by package name, with both builds of each (the desktop's
# from the product's link, the device's from the objects below, which rename every entry <entry>_device_mode).
set(_rp_decls "")
set(_rp_rows "")
set(_rp_count 0)
foreach(_p IN LISTS RP_EXPORT_PLAN_PACKAGES)
    get_property(_entries GLOBAL PROPERTY RP_PACKAGE_${_p}_DEVICE_ENTRIES)
    foreach(_e IN LISTS _entries)
        string(APPEND _rp_decls "extern \"C\" const rp_package* ${_e}(void);\n"
                                "extern \"C\" const rp_package* ${_e}_device_mode(void);\n")
        string(APPEND _rp_rows "        {\"${_p}\", &${_e}, &${_e}_device_mode},\n")
        math(EXPR _rp_count "${_rp_count} + 1")
    endforeach()
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/export-plan/export-plan-packages.cpp" CONTENT
"// Generated by packages/graph-runtime/export-plan/CMakeLists.txt from the manifests of RP_EXPORT_PLAN_PACKAGES: the
// device entries --packages chooses among, by package. Do not edit.

#include \"ExportPackages.hpp\"

${_rp_decls}
const RpExportPackage* rp_export_plan_packages(std::size_t* count) {
    static const RpExportPackage kPackages[] = {
${_rp_rows}    };
    *count = ${_rp_count};
    return kPackages;
}
")
add_executable(retroplug-export-plan EXCLUDE_FROM_ALL ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp
    "${CMAKE_BINARY_DIR}/export-plan/export-plan-packages.cpp")
target_include_directories(retroplug-export-plan PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(retroplug-export-plan PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
retroplug_product(retroplug-export-plan PACKAGES ${RP_EXPORT_PLAN_PACKAGES} DEVICE)
retroplug_device_mode_objects(retroplug-export-plan-device PACKAGES ${RP_EXPORT_PLAN_PACKAGES})
target_sources(retroplug-export-plan PRIVATE $<TARGET_OBJECTS:retroplug-export-plan-device>)
target_link_libraries(retroplug-export-plan PRIVATE retroplug-graph retroplug-node-sdk)
