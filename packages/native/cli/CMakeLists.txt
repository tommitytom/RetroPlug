# retroplug-cli: headless renderer + scripted-input runner.
#
# Reuses the DSP-side sources directly (Project.cpp + SameBoySystem.cpp);
# no DPF, no LVGL. Output binary lands in build/bin/.

# --- retroplug-cli-core: HarnessRpcService (the CLI / test-harness RPC surface) over retroplug-core.
# The shared emulator/Project core now lives in the neutrally-named `retroplug-core` lib (defined in the
# greenfield tree — packages/native-greenfield/CMakeLists.txt); this is the thin CLI-only wrapper that adds
# HarnessRpcService on top. Kept free of txiki/QuickJS (like retroplug-core) so harness-schema-dump links
# without the JS runtime — that decoupling is what breaks the build cycle once retroplug-cli embeds a TS
# bundle compiled from this schema.
add_library(retroplug-cli-core STATIC HarnessRpcService.cpp)

# HarnessRpcService.cpp's own headers (HarnessRpcService.hpp / HarnessRpcRegistration.hpp /
# TestHarnessImpl.hpp / SchemaDumpCodec.hpp) live in this dir; the shared-core includes + all the core
# libs (sameboy / mesen / rpcpp / …) arrive PUBLIC from retroplug-core.
target_include_directories(retroplug-cli-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(retroplug-cli-core PUBLIC retroplug-core)

set_target_properties(retroplug-cli-core PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

# --- retroplug-cli: the txiki host for the embedded CLI app + test harness.
# ${CLI_BUNDLE_C} (the end-user CLI bundled to QuickJS bytecode) is generated by
# cli-bundle-regenerate; expands to empty when Node is absent (then the default
# branch needs RETROPLUG_CLI_BUNDLE_PATH).
add_executable(retroplug-cli
    main.cpp
    TestHarness.cpp  # embedded QuickJS TypeScript test harness (--test)
    ${DPFJS_PATH}/src/dpfjs/host/TjsHostRuntime.cpp  # shared txiki host (dpf.js)
    ${CLI_BUNDLE_C}
)

if(CLI_BUNDLE_C)
    target_compile_definitions(retroplug-cli PRIVATE RETROPLUG_CLI_BUNDLE_EMBEDDED)
endif()

target_include_directories(retroplug-cli PRIVATE
    # dpf.js src root for the shared host header ("dpfjs/host/TjsHostRuntime.hpp").
    ${DPFJS_PATH}/src
    # txiki.js / QuickJS runtime headers for the embedded JS test harness.
    # `tjs.h` + internal `private.h` (TJS_EvalModuleContent / TJS_GetLoop /
    # tjs__execute_jobs); `quickjs.h` arrives transitively via the `tjs` link
    # but is listed for clarity.
    ${DPFJS_PATH}/deps/lv_binding_js/deps/txiki/src
    ${DPFJS_PATH}/deps/lv_binding_js/deps/txiki/deps/quickjs
)

target_link_libraries(retroplug-cli PRIVATE
    retroplug-cli-core
    tjs       # txiki.js/QuickJS runtime for the embedded TS test harness
              # (transitively brings qjs + libuv)
)

set_target_properties(retroplug-cli PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)

# --- harness-schema-dump: prints HarnessRpcService's OpenRPC schema for codegen.
# No txiki link — that's the whole point (see retroplug-cli-core above). Mirrors
# the plugin's rpc-schema-dump.
add_executable(harness-schema-dump HarnessSchemaDump.cpp)
target_link_libraries(harness-schema-dump PRIVATE retroplug-cli-core)
set_target_properties(harness-schema-dump PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)
